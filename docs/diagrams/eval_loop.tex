  node[sglabel, pos=0.5, below] {accept / edit / reject} (persona.south east);

\node[sgcaption, align=center, text width=44mm] at (33mm, 15mm)
  {the engine is never told\\ the hidden weights};

% ---------------- taps ----------------
\node[sgsmall, text width=36mm, align=left, below=17mm of persona, xshift=-2mm] (t1)
  {\textbf{acceptance} over decisions\\[-1pt]
   {\scriptsize with edit distance beside it, so\\[-1pt]
    "accepted more" cannot hide\\[-1pt] "rewrote every word"}};

    with the clean-story false rate}};
\node[sgsmall, text width=36mm, align=left, below=6mm of t3] (t4)

\node[sgsmall, text width=36mm, align=left, below=6mm of t4] (t5)
  {\textbf{diversity vs quality}, \textbf{regret},\\[-1pt]
   \textbf{tokens per action}};

\node[sgcaption, anchor=north west, text width=124mm] at (-22mm,-72mm)
  {The personas are defined over the same feature space the preference head is fitted on,
   which turns the central question into a measurable one: not ``did acceptance rise'' ---
   any degenerate system achieves that --- but ``did the machinery recover the taste it was
   shown''. What this cannot measure is real taste, fatigue, or trust. A persona is a
   linear functional plus noise; only a human study would settle those.};
